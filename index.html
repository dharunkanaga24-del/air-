ESP32 Firmware & Hardware Circuit Guide

🔌 Hardware Wiring Diagram

Connect your MQ Gas / Air Quality sensor (MQ-135, MQ-2, MQ-7, etc.) to the ESP32:

• Sensor VCC (Pin 1)  ➔  ESP32 5V (VIN) or 3.3V
• Sensor GND (Pin 2)  ➔  ESP32 GND
• Sensor AOUT (Pin 3) ➔  ESP32 GPIO 34 (Analog Input Pin - ADC1)
• Sensor DOUT (Pin 4) ➔  Leave disconnected (or connect to digital pin for threshold interrupt)

📄 Arduino C++ Code (Ready to Flash)

#include <WiFi.h>
#include <Firebase_ESP_Client.h>

// 1. Put your Wi-Fi credentials here
#define WIFI_SSID "YOUR_WIFI_NAME"
#define WIFI_PASSWORD "YOUR_WIFI_PASSWORD"

// 2. Put your Firebase credentials here
#define API_KEY "YOUR_FIREBASE_API_KEY"
#define DATABASE_URL "YOUR_DATABASE_URL" 

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

const int sensorPin = 34; // Analog pin GPIO 34

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nConnected!");

  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;
  config.signer.test_mode = true;
  
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  int airQualityReading = analogRead(sensorPin);
  
  if (Firebase.ready()) {
    if (Firebase.RTDB.setInt(&fbdo, "sensor/air_quality", airQualityReading)) {
      Serial.println("Data sent: " + String(airQualityReading));
    } else {
      Serial.println("Failed to send: " + fbdo.errorReason());
    }
  }
  
  delay(2000); // 2 second update loop
}
📦 Required Arduino Libraries: Open Arduino IDE > Sketch > Include Library > Manage Libraries > Search and install Firebase ESP Client by Mobizt.
